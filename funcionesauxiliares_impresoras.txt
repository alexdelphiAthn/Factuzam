uses
  StrUtils, SysUtils, Printers;

// 1. Función auxiliar para quitar tildes y diéresis
function QuitarAcentos(const Texto: string): string;
const
  // Mapeo simple: Origen -> Destino
  ConAcento = 'áéíóúüñçÁÉÍÓÚÜÑÇàèìòùÀÈÌÒÙäëïöüÄËÏÖÜâêîôûÂÊÎÔÛ';
  SinAcento = 'aeiouuncAEIOUUNCaeiouAEIOUaeiouAEIOUaeiouAEIOU';
var
  i: Integer;
begin
  Result := Texto;
  for i := 1 to Length(ConAcento) do
    Result := StringReplace(Result, ConAcento[i], SinAcento[i], [rfReplaceAll]);
end;

// 2. Función principal de búsqueda
function BuscarImpresora(TextoParcial: string): string;
var
  i: Integer;
  NombreLimpioImp, NombreLimpioBusqueda: string;
begin
  Result := '';
  if Trim(TextoParcial) = '' then Exit;

  // Limpiamos la búsqueda (quitamos tildes)
  // Nota: No hace falta hacer UpperCase aquí si usamos ContainsText luego
  NombreLimpioBusqueda := QuitarAcentos(TextoParcial);

  for i := 0 to Printer.Printers.Count - 1 do
  begin
    // Limpiamos el nombre de la impresora real (quitamos tildes)
    NombreLimpioImp := QuitarAcentos(Printer.Printers[i]);

    // AHORA SÍ: Usamos ContainsText.
    // ContainsText ignora mayúsculas/minúsculas automáticamente.
    // Como ya hemos quitado los acentos manualmente antes, ahora coincide todo.
    if ContainsText(NombreLimpioImp, NombreLimpioBusqueda) then
    begin
      Result := Printer.Printers[i]; // Devolvemos el nombre ORIGINAL
      Exit;
    end;
  end;
end;

uses

function NormalizarTexto(const Texto: string): string;
const
  // Lista completa de caracteres "problemáticos" en español y europeos
  ConAcento = 'áéíóúüñçÁÉÍÓÚÜÑÇàèìòùÀÈÌÒÙäëïöüÄËÏÖÜâêîôûÂÊÎÔÛ';
  SinAcento = 'aeiouuncAEIOUUNCaèiòùAEIOUUaeiouuAEIOUUaeiouAEIOU';
var
  i: Integer;
begin
  Result := Texto;
  // 1. Recorremos la lista de caracteres raros
  for i := 1 to Length(ConAcento) do
  begin
    // 2. Reemplazamos todas las ocurrencias en un solo paso
    //    Si encuentras 'ü', cámbialo por 'u'.
    Result := StringReplace(Result, ConAcento[i], SinAcento[i], [rfReplaceAll]);
  end;
  
  // 3. Convertimos a mayúsculas para asegurar el match final
  Result := UpperCase(Result); 
end;
