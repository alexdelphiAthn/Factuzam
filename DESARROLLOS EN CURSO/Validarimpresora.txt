procedure TForm1.rowImpresoraTicketsEditPropertiesValidate(
  Sender: TObject; var DisplayValue: Variant; var ErrorText: TCaption;
  var Error: Boolean);
var
  TextoUsuario: string;
  ImpresoraEncontrada: string;
begin
  TextoUsuario := VarToStr(DisplayValue);

  // Si lo dejan vacío, no validamos nada (asumimos que quieren borrar)
  if Trim(TextoUsuario) = '' then Exit;

  // Buscamos si existe alguna impresora que coincida
  ImpresoraEncontrada := BuscarCoincidenciaImpresora(TextoUsuario);

  if ImpresoraEncontrada <> '' then
  begin
    // CASO 1: Encontrada.
    // Preguntamos al usuario si quiere confirmar esa coincidencia.
    if MessageDlg('Se ha encontrado una impresora activa que coincide:' + sLineBreak +
                  sLineBreak +
                  '--> ' + ImpresoraEncontrada + sLineBreak +
                  sLineBreak +
                  '¿Es correcto?', mtConfirmation, [mbYes, mbNo], 0) = mrYes then
    begin
      // OPCIONAL: Si quieres que se guarde el nombre PARCIAL (ej: "HP"), no hagas nada aquí.
      // OPCIONAL: Si prefieres que se autocomplete con el nombre REAL detectado, descomenta esto:
      // DisplayValue := ImpresoraEncontrada; 
    end
    else
    begin
      // El usuario dijo que NO era esa.
      ErrorText := 'Impresora rechazada por el usuario.';
      Error := True; // Esto mantiene el foco en la celda y muestra el icono de error
    end;
  end
  else
  begin
    // CASO 2: No existe ninguna impresora con ese nombre parcial.
    ErrorText := 'ATENCIÓN: No se detecta ninguna impresora instalada que contenga el texto "' + TextoUsuario + '".' +
                 sLineBreak + 'Verifique el nombre o instale el driver.';
    Error := True; // Bloquea la salida hasta que lo arreglen o borren
  end;
end;
